<script>
(() => {
  const dialog = document.getElementById('pub-bibtex-dialog');
  const entries = document.getElementById('pub-bibtex-entries').content;
  const bibtex = document.getElementById('pub-bibtex-text');
  const status = document.getElementById('pub-bibtex-status');

  document.querySelectorAll('.pub-cite[data-bibtex]').forEach((button) => {
    button.addEventListener('click', () => {
      bibtex.value = entries.getElementById('bibtex-' + button.dataset.bibtex).textContent.trim();
      status.textContent = '';
      dialog.showModal();
    });
  });
  document.getElementById('pub-bibtex-close').addEventListener('click', () => dialog.close());
  document.getElementById('pub-bibtex-copy').addEventListener('click', async () => {
    try {
      await navigator.clipboard.writeText(bibtex.value);
      status.textContent = 'Copied';
    } catch {
      bibtex.focus();
      bibtex.select();
      status.textContent = 'Press Ctrl+C or ⌘C to copy.';
    }
  });

  const badges = document.querySelectorAll('.pub-scholar[data-scholar-title]');
  const profileId = '4CxuLpsAAAAJ';
  const feeds = [
    'https://raw.githubusercontent.com/wanxinhang/wanxinhang.github.io/google-scholar-stats/gs_data.json',
    'https://cdn.jsdelivr.net/gh/wanxinhang/wanxinhang.github.io@google-scholar-stats/gs_data.json'
  ];
  const normalizeTitle = (title) => title.normalize('NFKC').toLowerCase().replace(/[^a-z0-9]/g, '');
  const numberFormat = new Intl.NumberFormat('en-US');
  let refreshing = false;
  let hasData = false;

  function renderCitations(data) {
    const papers = new Map();
    for (const [id, paper] of Object.entries(data.publications)) {
      const title = paper?.bib?.title;
      const count = paper?.num_citations;
      if (!id.startsWith(profileId + ':') || typeof title !== 'string' || !Number.isInteger(count) || count < 0) continue;

      const key = normalizeTitle(title);
      const existing = papers.get(key);
      if (!existing || count > existing.count) papers.set(key, { id, count });
    }
    if (!papers.size) return false;

    for (const badge of badges) {
      const paper = papers.get(normalizeTitle(badge.dataset.scholarTitle));
      if (!paper) continue;

      badge.querySelector('.pub-scholar-count').textContent = numberFormat.format(paper.count);
      const url = new URL('https://scholar.google.com/citations');
      url.searchParams.set('view_op', 'view_citation');
      url.searchParams.set('user', profileId);
      url.searchParams.set('citation_for_view', paper.id);
      badge.href = url.href;
      badge.title = data.updated ? 'Google Scholar · Last sync: ' + data.updated : 'View this paper on Google Scholar';
    }
    return true;
  }

  async function refreshCitations() {
    if (refreshing || document.hidden) return;

    refreshing = true;
    try {
      for (const url of feeds) {
        try {
          const response = await fetch(url, { cache: 'no-store', signal: AbortSignal.timeout(6000) });
          if (!response.ok) continue;

          const data = await response.json();
          if (data.scholar_id === profileId && data.publications && renderCitations(data)) {
            hasData = true;
            return;
          }
        } catch {
          // The second feed can serve the latest published data if the first is unavailable.
        }
      }
      if (!hasData) {
        badges.forEach((badge) => {
          badge.title = 'Citation count unavailable. Open Google Scholar to view this paper.';
        });
      }
    } finally {
      refreshing = false;
    }
  }

  refreshCitations();
  window.setInterval(refreshCitations, 5 * 60 * 1000);
  document.addEventListener('visibilitychange', refreshCitations);
})();
</script>
